import asyncio, json, os
from pathlib import Path
from sqlalchemy import text, event
from sqlalchemy.ext.asyncio import create_async_engine
from app.database import _ensure_pg_trgm_indexes
from app.services.search_text_schema import ensure_search_text_columns
TABLES=('audio_works','movies','tv_series','work_collections')
async def snapshot(conn):
    return {t:[list(r) for r in (await conn.execute(text(f'SELECT id, search_text, aliases FROM {t} ORDER BY id'))).all()] for t in TABLES}
async def types(conn):
    return dict((await conn.execute(text("SELECT table_name,data_type FROM information_schema.columns WHERE table_schema=current_schema() AND column_name='search_text'"))).all())
async def main():
    e=create_async_engine(os.environ['DATABASE_URL']); statements=[]
    event.listen(e.sync_engine,'before_cursor_execute',lambda c,cur,s,p,ctx,m: statements.append(s))
    try:
        async with e.begin() as c:
            await _ensure_pg_trgm_indexes(c)
            before=await snapshot(c)
            original=await types(c)
            assert all(original[t]=='character varying' for t in TABLES)
            indexes=(await c.execute(text("SELECT indexname,indexdef FROM pg_indexes WHERE schemaname=current_schema() AND indexname LIKE '%search_text_trgm' ORDER BY indexname"))).all()
        try:
            async with e.begin() as c:
                await ensure_search_text_columns(c)
                assert all((await types(c))[t]=='text' for t in TABLES)
                raise RuntimeError('injected rollback')
        except RuntimeError as error:
            assert str(error)=='injected rollback'
        async with e.begin() as c:
            assert await types(c)==original
            assert await snapshot(c)==before
            await ensure_search_text_columns(c)
        async with e.begin() as c:
            assert await snapshot(c)==before
            assert all((await types(c))[t]=='text' for t in TABLES)
            statements.clear()
            await ensure_search_text_columns(c)
            assert not any(s.startswith('ALTER TABLE') for s in statements)
            assert indexes==(await c.execute(text("SELECT indexname,indexdef FROM pg_indexes WHERE schemaname=current_schema() AND indexname LIKE '%search_text_trgm' ORDER BY indexname"))).all()
        result={'rollback_preserves_types_and_rows':True,'upgrade_preserves_rows':True,'repeat_no_ddl':True,'gin_definitions_preserved':len(indexes),'row_counts':{t:len(r) for t,r in before.items()}}
        Path('/tmp/rssripple-v33-upgrade-c.json').write_text(json.dumps(result,indent=2)+'\n'); print(result)
    finally: await e.dispose()
asyncio.run(main())
