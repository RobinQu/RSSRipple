"""Actual model defaults and HTTP serialization under explicit test time zones."""
import hashlib
import json
import os
import uuid
from pathlib import Path

import httpx
import pytest
from fastapi import FastAPI
from sqlalchemy import insert, select, text, update
from sqlalchemy.ext.asyncio import async_sessionmaker, create_async_engine

import app.models  # noqa: F401
from app.api.v1.channels import router
from app.database import Base, get_db
from app.models.channel import Channel
from app.utils.time import utcnow

CORPUS = Path('/workspace/dropbox/github/robinqu/RSSRipple/tests/fixtures/prod_works_v1.json')


@pytest.mark.asyncio
@pytest.mark.parametrize('zone', ['UTC', 'Asia/Shanghai', 'America/New_York'])
@pytest.mark.parametrize('boundary', ['created', 'updated', 'http'])
async def test_utc_contract(zone, boundary, record_property):
    raw = CORPUS.read_bytes()
    assert hashlib.sha256(raw).hexdigest() == 'd11651d2162ced23e8d919af0bff2d9f316e203234cc854909ba5f444a35ec32'
    recorded = json.loads(raw)['tables']['channels'][0]
    engine = create_async_engine(os.environ['TIME_PROBE_URL'], connect_args={'server_settings': {'timezone': zone}})
    factory = async_sessionmaker(engine, expire_on_commit=False)
    key = str(uuid.uuid4())
    try:
        async with engine.begin() as conn:
            await conn.run_sync(Base.metadata.create_all)
            assert await conn.scalar(text('SHOW TIMEZONE')) == zone
            await conn.execute(insert(Channel).values(
                id=key, name=recorded['name'], url=recorded['url'],
                field_mapping=recorded['field_mapping'], last_fetched_at=utcnow(),
            ))
        async with engine.begin() as conn:
            if boundary == 'updated':
                await conn.execute(update(Channel).where(Channel.id == key).values(name=recorded['name']+' [synthetic update]'))
            reference = await conn.scalar(text("SELECT CURRENT_TIMESTAMP AT TIME ZONE 'UTC'"))
            row = (await conn.execute(select(Channel.__table__).where(Channel.id == key))).mappings().one()
        field = 'updated_at' if boundary == 'updated' else 'created_at'
        offset = (row[field]-reference).total_seconds()
        python_offset = (row['last_fetched_at']-reference).total_seconds()
        record_property('zone', zone)
        record_property('database_offset_seconds', offset)
        record_property('python_offset_seconds', python_offset)
        assert abs(python_offset) < 5
        if boundary == 'http':
            app = FastAPI()
            app.include_router(router, prefix='/api/v1')
            async def session():
                async with factory() as db:
                    yield db
            app.dependency_overrides[get_db] = session
            async with httpx.AsyncClient(transport=httpx.ASGITransport(app=app), base_url='http://synthetic') as client:
                response = await client.get('/api/v1/channels/'+key)
            assert response.status_code == 200,response.text
            data = response.json()['data']
            record_property('created_at_json', data['created_at'])
            record_property('last_fetched_at_json', data['last_fetched_at'])
            assert data['created_at'].endswith(('Z','+00:00'))
            assert data['last_fetched_at'].endswith(('Z','+00:00'))
        else:
            assert abs(offset) < 5, (zone,boundary,offset)
    finally:
        await engine.dispose()
