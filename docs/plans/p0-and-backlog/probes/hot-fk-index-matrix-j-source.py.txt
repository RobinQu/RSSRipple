async def measure(engine,qs):
 async with engine.begin() as conn:await conn.execute(text('ANALYZE'))
 result={}
 async with engine.connect() as conn:
  for name,query in qs.items():
   sql=str(query.compile(dialect=engine.dialect,compile_kwargs={'literal_binds':True}))
   prefix='EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) ' if engine.dialect.name=='postgresql' else 'EXPLAIN QUERY PLAN '
   plan=[list(row) for row in (await conn.execute(text(prefix+sql))).all()]
   readings=[];identities=[]
   for _ in range(3):
    start=time.perf_counter();rows=(await conn.execute(query)).mappings().all();readings.append(time.perf_counter()-start)
    current=sorted(row['id'] for row in rows)
    if identities:assert current==identities
    identities=current
   result[name]={'sql':sql,'plan':plan,'seconds':readings,'count':len(identities),'result_ids_sha256':sha(json.dumps(identities).encode()),'result_ids':identities,'sample_rows':[{'id':row['id'],'created_at':str(row.get('created_at')),'decision_key':row.get('decision_key'),'agent_id':row.get('agent_id')} for row in rows[:30]]}
 return result

async def probe(url):

 try:
  async with engine.begin() as conn:
   await conn.run_sync(Base.metadata.create_all)
   if engine.dialect.name=='sqlite':await conn.execute(text('PRAGMA journal_mode=mvcc'))
  ids=await seed(engine);qs=queries(ids);before=await measure(engine,qs)
  specs=[(FileResource,('movie_id',)),(FileResource,('series_id',)),(FileResource,('collection_id',)),(AgentWork,('agent_id',)),(AgentWork,('movie_id',)),(AgentWork,('series_id',)),(PendingDecision,('agent_id','created_at')),(PendingDecision,('movie_id',)),(PendingDecision,('series_id',)),(DownloadTask,('file_resource_id',)),(DownloadTask,('agent_id',)),(AgentRun,('agent_id','started_at')),(WebhookDelivery,('status','next_attempt_at','created_at'))]
  indexes=[]
  async with engine.begin() as conn:
   for model,columns in specs:
    index=Index('ix_probe_'+model.__tablename__+'_'+'_'.join(columns),*[model.__table__.c[col] for col in columns]);await conn.run_sync(index.create);indexes.append(index)
  after=await measure(engine,qs)
  mismatches=[k for k in qs if before[k]['result_ids_sha256']!=after[k]['result_ids_sha256']]
  Path(os.environ['INDEX_PROBE_REPORT']).write_text(json.dumps({'before':before,'after':after,'mismatches':mismatches},ensure_ascii=False,indent=2,default=str)+'\n')
  assert not mismatches,mismatches
  # Compare alternative delivery order index independently, not with both
  # present and a planner choice mistaken for isolated evidence.
  async with engine.begin() as conn:
   await conn.run_sync(indexes[-1].drop)
   index=Index('ix_probe_webhook_status_created',WebhookDelivery.__table__.c.status,WebhookDelivery.__table__.c.created_at);await conn.run_sync(index.create)
  alternative=await measure(engine,{'webhook_due':qs['webhook_due']})
  assert alternative['webhook_due']['result_ids_sha256']==before['webhook_due']['result_ids_sha256']
  async with engine.begin() as conn:
   await conn.run_sync(index.drop)
   await conn.execute(update(WebhookDelivery).values(status='pending',next_attempt_at=case(
    (WebhookDelivery.created_at<=NOW-timedelta(seconds=N-8),None),

    else_=NOW+timedelta(days=1))))
  future_heavy={'none':await measure(engine,{'webhook_due':qs['webhook_due']})}
  for name,cols in [('status_next_created',('status','next_attempt_at','created_at')),('status_created',('status','created_at'))]:
   ix=Index('ix_probe_future_'+name,*[WebhookDelivery.__table__.c[col] for col in cols])
   async with engine.begin() as conn:await conn.run_sync(ix.create)
   future_heavy[name]=await measure(engine,{'webhook_due':qs['webhook_due']})
   async with engine.begin() as conn:await conn.run_sync(ix.drop)
  expected=future_heavy['none']['webhook_due']['result_ids_sha256']
  assert all(stage['webhook_due']['count']==16 and stage['webhook_due']['result_ids_sha256']==expected for stage in future_heavy.values())
  return {'future_heavy_8192_pending_16_due':future_heavy,'backend':engine.dialect.name,'corpus_sha256':ids['corpus_sha256'],'rows_per_hot_table':N,'agents':M,'works_per_agent':8,'before':before,'after':after,'alternative_webhook_status_created':alternative,'scope':'Recorded titles; UUIDs, assignments, timestamps, payloads and distributions synthetic.  finally:await engine.dispose()
